cmake_minimum_required(VERSION 3.16)
project(colab_systemc_examples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard for the examples and fetched SystemC")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(COLAB_FETCH_SYSTEMC "Download/build pinned SystemC instead of using an installed package" OFF)
option(COLAB_COMPILER_SMOKE_TEST "Build and execute an RV32 C program (requires Clang, LLD and Python)" OFF)
option(COLAB_RTL_TESTS "Run RTL simulations and synthesis (requires Icarus, Yosys, Clang, LLD and Python)" OFF)

if(COLAB_FETCH_SYSTEMC)
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
    set(ENABLE_EXAMPLES OFF CACHE BOOL "Build SystemC's own examples" FORCE)
    set(ENABLE_REGRESSION OFF CACHE BOOL "Build SystemC's own regression suite" FORCE)
    FetchContent_Declare(systemc
        URL https://codeload.github.com/accellera-official/systemc/tar.gz/refs/tags/3.0.2
        URL_HASH SHA256=9b3693ed286aab958b9e5d79bb0ad3bc523bbc46931100553275352038f4a0c4
    )
    FetchContent_MakeAvailable(systemc)
    # Keep warnings in our code strict; older SystemC TLM headers also trigger
    # overloaded-virtual diagnostics when consumed as ordinary source headers.
    get_target_property(COLAB_SYSTEMC_HEADERS systemc INTERFACE_INCLUDE_DIRECTORIES)
    set_property(TARGET systemc APPEND PROPERTY INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${COLAB_SYSTEMC_HEADERS}")
else()
    find_package(SystemCLanguage CONFIG QUIET NO_CMAKE_PACKAGE_REGISTRY)
    if(NOT TARGET SystemC::systemc)
        # Distribution packages such as libsystemc-dev may omit the CMake config.
        find_path(COLAB_SYSTEMC_INCLUDE_DIR NAMES systemc)
        find_library(COLAB_SYSTEMC_LIBRARY NAMES systemc)
        if(NOT COLAB_SYSTEMC_INCLUDE_DIR OR NOT COLAB_SYSTEMC_LIBRARY)
            message(FATAL_ERROR
                "SystemC not found. Set CMAKE_PREFIX_PATH to its installation, "
                "install libsystemc-dev, or configure with -DCOLAB_FETCH_SYSTEMC=ON.")
        endif()
        find_package(Threads REQUIRED)
        add_library(colab_systemc INTERFACE)
        target_include_directories(colab_systemc SYSTEM INTERFACE "${COLAB_SYSTEMC_INCLUDE_DIR}")
        target_link_libraries(colab_systemc INTERFACE "${COLAB_SYSTEMC_LIBRARY}" Threads::Threads)
        add_library(SystemC::systemc ALIAS colab_systemc)
    endif()
endif()

# Match the library's C++ standard; its API version check also checks this ABI.
if(DEFINED SystemC_CXX_STANDARD)
    if(SystemC_CXX_STANDARD LESS 17 OR SystemC_CXX_STANDARD EQUAL 98)
        message(FATAL_ERROR "Rebuild SystemC with CMAKE_CXX_STANDARD=17 or newer.")
    endif()
    set(CMAKE_CXX_STANDARD "${SystemC_CXX_STANDARD}")
endif()

    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(systemc_${example} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
